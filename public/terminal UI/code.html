# Lesson 5: Loops & Logic
# Iterate through a list of server nodes
 
nodes = ["Alpha", "Beta", "Gamma", "Delta"]
 
def ping_servers(server_list):
    print("Initiating connection sequence...")
 
    # Loop through each node
    for node in server_list:
        status = "Online"
        if node == "Gamma":
            status = "Offline"
        print(f"Node: {node} | Status: {status}")
 
ping_servers(nodes)
